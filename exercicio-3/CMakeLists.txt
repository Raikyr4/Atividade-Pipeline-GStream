cmake_minimum_required(VERSION 3.20)
project(exercicio_3_streaming LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(exercicio_3 src/principal.cpp)
if(MSVC)
    set(GSTREAMER_RAIZ "$ENV{GSTREAMER_1_0_ROOT_MSVC_X86_64}" CACHE PATH "SDK GStreamer x64")
    find_path(GST_INCLUDE gst/gst.h PATHS "${GSTREAMER_RAIZ}/include/gstreamer-1.0" REQUIRED)
    find_path(GLIB_INCLUDE glib.h PATHS "${GSTREAMER_RAIZ}/include/glib-2.0" REQUIRED)
    find_path(GLIB_CONFIG glibconfig.h PATHS "${GSTREAMER_RAIZ}/lib/glib-2.0/include" REQUIRED)
    target_include_directories(exercicio_3 PRIVATE "${GST_INCLUDE}" "${GLIB_INCLUDE}" "${GLIB_CONFIG}")
    foreach(LIB gstreamer-1.0 gobject-2.0 glib-2.0)
        find_library(LIB_${LIB} NAMES ${LIB} PATHS "${GSTREAMER_RAIZ}/lib" REQUIRED)
        target_link_libraries(exercicio_3 PRIVATE "${LIB_${LIB}}")
    endforeach()
    target_compile_options(exercicio_3 PRIVATE /W4 /utf-8)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.18)
    target_link_libraries(exercicio_3 PRIVATE PkgConfig::GSTREAMER)
    target_compile_options(exercicio_3 PRIVATE -Wall -Wextra -Wpedantic)
endif()
